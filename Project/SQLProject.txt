create database Library_Database_Managemnt_System ;

create table libraryStaff (
ID int primary key,
Name varchar(50),
ContactInfo varchar(50),
Section varchar(50),
EmpDate Date 
);

create table Members(
ID int primary key,
Name varchar(50) not null,
ConactInfo varchar(50),
Membership varchar(20),
RegesterDate Date
);

create table category (
ID int primary key,
Name varchar(50),
Cdescription varchar(500)
);

create table Books (
ID int primary key,
Title varchar(100) not null,
Author varchar(50) not null,
Genre varchar(50),
Publication Date,
AvailableStatus varchar(10),
CategoryID int,
Foreign key (CategoryID) references Category (ID)
);

create table Borrwoing(
ID int primary key,
BorrowingDate Date,
dueDate Date,
returnDate Date,
MemberID int,
BookID int ,
Foreign key (MemberID) references Members (ID),
Foreign key (BookID) references Books (ID) 
);

create table Reservation (
ID int primary key,
ReservationDate Date,
ReserveStatus varchar(15),
MemberID int,
BookID int ,
Foreign key (MemberID) references Members (ID),
Foreign key (BookID) references Books (ID)
);

create table Fines(
ID int primary key,
Amount float,
PaymantStatus varchar(10),
BorrowingID int unique,
Foreign key (BorrowingID) references Borrwoing (ID)
);
